set(NUM_OF_REPATING_OPS "1" "2" "4" "8" "16")
foreach(NUM_NOPS ${NUM_OF_REPATING_OPS})
    add_ubench(TARGET_NAME nops_${NUM_NOPS} SOURCE_FILE nops.cc DEFINITIONS ARRAY_SIZE=${NUM_NOPS})
endforeach()

set(ARITHMETIC_OPS "add" "sub")
foreach(OP ${ARITHMETIC_OPS})
    foreach(NUM_OPS ${NUM_OF_REPATING_OPS})
        add_ubench(TARGET_NAME arithmetic_${OP}_${NUM_OPS} SOURCE_FILE arithmetic.cc DEFINITIONS OP=${OP} ARRAY_SIZE=${NUM_OPS})
    endforeach()
endforeach()
